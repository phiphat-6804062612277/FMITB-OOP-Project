Class Diagram: Five Minutes In The Bathroom (รูปแบบเรียงแนวดิ่ง A4)

    direction TB

    %% ---------------------------------------------------------
    %% LAYER 1: ENTRY POINT & WINDOWING
    %% ---------------------------------------------------------
    class Main {
        +main(String args) void
    }

    class GameWindow {
        -GamePanel gamePanel
        +GameWindow()
        +showWindow() void
    }

    class GamePanel {
        +int PANEL_WIDTH
        +int PANEL_HEIGHT
        +int TARGET_FPS
        -Thread gameThread
        -boolean isRunning
        -GameTimer gameTimer
        -StaminaManager staminaManager
        -BathroomManager bathroomManager
        -CameraManager cameraManager
        -MonsterAI monsterAI
        -StateManager stateManager
        -MenuRenderer menuRenderer
        -HowToPlayRenderer howToPlayRenderer
        -BathroomRenderer bathroomRenderer
        -TabletRenderer tabletRenderer
        -EndingRenderer gameOverRenderer
        -EndingRenderer gameWinRenderer
        +startGameLoop() void
        +stopGameLoop() void
        +run() void
        #paintComponent(Graphics g) void
        +mousePressed(MouseEvent e) void
        +keyPressed(KeyEvent e) void
    }

    Main ..> GameWindow : 1. launches
    GameWindow *-- GamePanel : 2. embeds

    %% ---------------------------------------------------------
    %% LAYER 2: STATE MANAGEMENT
    %% ---------------------------------------------------------
    class GameState {
        +int MENU
        +int HOW_TO_PLAY
        +int BATHROOM
        +int TABLET
        +int GAME_OVER
        +int GAME_WIN
    }

    class StateManager {
        -int currentState
        -GameTimer timer
        -StaminaManager stamina
        -BathroomManager bathroom
        -CameraManager camera
        -MonsterAI monsterAI
        +StateManager(GameTimer timer, StaminaManager stamina, BathroomManager bathroom, CameraManager camera, MonsterAI monsterAI)
        +startNewGame() void
        +setState(int newState) void
        +getCurrentState() int
        +isPlaying() boolean
    }

    GamePanel *-- StateManager : 3. controls
    StateManager ..> GameState : uses constants

    class CameraManager {
        -CameraId activeCamera

        +reset() void

        +toggleFloor() void
        +setFloor(int floor) void

        +getActiveFloor() int
    }

    class MonsterAI {
        -MonsterArray monsters
        -SahurMonster sahur
        -TralalaMonster tralala
        +MonsterAI()
        +reset() void
        +update(float dt, float gameProgress, BathroomManager bathroom) boolean
        +getSahur() SahurMonster
        +getTralala() TralalaMonster
    }

    class Monster {
        <>
        #String name
        #int stage
        #int maxStage
        #float moveTimer
        #float baseMoveInterval
        #float attackCountdown
        #float baseAttackReactionTime
        #float defenseHoldTimer
        #float requiredHoldTime
        #boolean isAtAttackPoint
        #boolean hasKilledPlayer
        +Monster(String name, int maxStage, float baseMoveInterval, float baseAttackReactionTime, float requiredHoldTime)
        +reset() void
        +updateAI(float dt, float gameProgress, BathroomManager bathroom) boolean
        +onBlocked(BathroomManager bathroom) void
        +getName() String
        +getStage() int
        +isAtAttackPoint() boolean
    }

    class SahurMonster {
        +SahurMonster()

    }

    class TralalaMonster {
        +TralalaMonster()

    class AssetManager {

    }

    class SoundManager {
        -SoundManager instance
        -StringArray clipNames

        -float masterVolume
        +getInstance() SoundManager

        +stopSound(String soundName) void

    StateManager o-- GameTimer
    StateManager o-- StaminaManager
    StateManager o-- BathroomManager
    StateManager o-- CameraManager
    StateManager o-- MonsterAI

    BathroomManager o-- StaminaManager : controls drain
    CameraManager o-- CameraId : selects
    MonsterAI *-- Monster : polymorphically manages
    SahurMonster --|> Monster : inherits
    TralalaMonster --|> Monster : inherits
